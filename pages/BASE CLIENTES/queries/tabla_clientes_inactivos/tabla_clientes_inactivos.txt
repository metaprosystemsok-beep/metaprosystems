SELECT 
    ROW_NUMBER() OVER () AS id,
    MAX(nombre_cliente) AS nombre_cliente,
    email,
    MAX(telefono) AS telefono,
    MAX(ultima_compra) AS ultima_compra,
    EXTRACT(DAY FROM NOW() - MAX(ultima_compra))::INT AS dias_inactivo,
    '' AS mensaje_enviado_mail,
    '' AS mensaje_enviado_whatsapp
FROM public.base_clientes
WHERE email NOT IN (
    SELECT email FROM public.clientes_inactivos WHERE email IS NOT NULL
)
GROUP BY email
HAVING MAX(ultima_compra) < NOW() - INTERVAL '45 days'
ORDER BY ultima_compra ASC;
