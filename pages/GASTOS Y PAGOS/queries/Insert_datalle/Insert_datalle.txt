SELECT
  to_char(CAST(fecha AS DATE), 'YYYY-MM-DD') AS fecha,
  detalle,
  total
FROM public.tabla_detalles
WHERE
  (
    COALESCE(TRIM('{{ Select2.selectedOptionValue || "" }}'), '') <> ''
    OR COALESCE(TRIM('{{ Select1.selectedOptionValue || "" }}'), '') <> ''
  )
  AND (
    COALESCE(TRIM('{{ Select1.selectedOptionValue || "" }}'), '') = ''
    OR TRIM(detalle) = TRIM('{{ Select1.selectedOptionValue || "" }}')
  )
  AND (
    '{{ DatePicker7.formattedDate || "" }}' = ''
    OR CAST(fecha AS DATE) >= CAST('{{ DatePicker7.formattedDate || "" }}' AS DATE)
  )
  AND (
    '{{ DatePicker6.formattedDate || "" }}' = ''
    OR CAST(fecha AS DATE) <= CAST('{{ DatePicker6.formattedDate || "" }}' AS DATE)
  )
ORDER BY CAST(fecha AS DATE) DESC
LIMIT {{ Tabla_detalle.pageSize || 10 }}
OFFSET {{ ((Tabla_detalle.pageNo || 1) - 1) * (Tabla_detalle.pageSize || 10) }};
